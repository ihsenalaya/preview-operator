  Quick start — create your first preview env:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  kubectl apply -f - <<EOF
  apiVersion: platform.company.io/v1alpha1
  kind: Preview
  metadata:
    name: pr-42
  spec:
    branch: feature/my-feature
    prNumber: 42
    image: ghcr.io/myorg/myapp:sha-abc
    ttl: 48h
    database:
      enabled: true
    aiEnrichment:
      enabled: true
      apiSecretRef:
        name: ai-api-key
        key: api-key
    testSuite:
      enabled: true
      smoke: {}
      regression:
        enabled: true
  EOF

  kubectl get prev          # shortname for Preview
  kubectl get prev pr-42 -o jsonpath='{.status.phase}'
  kubectl get prev pr-42 -o jsonpath='{.status.url}'

{{- if .Values.webhook.enabled }}
  ✓ Webhooks enabled — cert-manager must be installed.
{{- else }}
  ℹ Webhooks disabled (webhook.enabled=false) — no cert-manager required.
    Note: admission validation (large tier gate, Go telemetry checks) is skipped.
{{- end }}
  • Ingress controller required for preview URLs (ingress-nginx or Istio — auto-detected).
  • OpenTelemetry Operator optional — needed only for spec.telemetry.autoInstrumentation.
  • Microcks optional — needed only for spec.testSuite.contractTesting.
  • kagent optional — needed only for spec.kagent.enabled and spec.testStrategy.mode=Auto.

  kubectl create secret generic ai-api-key \
    --namespace {{ .Release.Namespace }} \
    --from-literal=api-key="<OPENAI_OR_AZURE_KEY>"

  # GitHub integration (required if spec.github.enabled)
  kubectl create secret generic preview-github-token \
    --namespace {{ .Release.Namespace }} \
    --from-literal=token="<GITHUB_PAT>"
